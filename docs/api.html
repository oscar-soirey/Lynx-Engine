A compact reference of the API actually registered in the engine's QuickJS runtime.

Actor Methods

getProperty(name)Read a reflected property.
setProperty(name, value)Modify a reflected property.
hasProperty(name)Check whether a property exists.
destroy()Request Actor destruction.
addTag(tag)Add a tag.
removeTag(tag)Remove a tag.
hasTag(tag)Check a tag.
addScript(path)Attach a script.
removeScript(path)Remove a script.
hasScript(path)Check for an attached script.
call(name, ...args)Call a function in the Actor's scripts.
toString()Return a text representation of the Actor.

Complete example

// scripts/Enemy.js
function BeginPlay() {
  parent.addTag("enemy");
  parent.lifetime = 20;
}

function Update(dt) {
  const player = Level.find("Player");
  if (!player || !player.valid)
    return;

  const direction = player.position.x > parent.position.x ? 1 : -1;
  parent.velocity.x = direction * 2;

  if (Input.pressed("attack"))
    player.call("TakeDamage", 10);
}

function EndPlay() {
  console.log("Enemy removed");
}
